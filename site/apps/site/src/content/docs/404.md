# `_headers` can only match the URL that was *requested*, so its
# `/404  X-Robots-Tag: noindex` rule covers a direct hit on /404/ and nothing
# else — a genuine not-found response is served for some other path entirely and
# never sees that rule. The meta tag travels with the body, so it covers both.
head:
  - tag: meta
    attrs:
      name: robots
      content: noindex
